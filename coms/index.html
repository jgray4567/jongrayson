const MEMORY_GH='https://raw.githubusercontent.com/jgray4567/jongrayson/main/memory/';
const overlay=document.getElementById('modal-overlay');
const mTitle=document.getElementById('modal-title');
const mContent=document.getElementById('modal-content');
const mLoading=document.getElementById('modal-loading');
const mPath=document.getElementById('modal-path');
const mLines=document.getElementById('modal-lines');

function openModal(title,path){
  mTitle.textContent=title;
  mContent.style.display='none';
  mLoading.style.display='block';
  mLoading.textContent='Fetching file...';
  overlay.classList.add('open');
  document.body.style.overflow='hidden';
  mPath.textContent=path;
  mLines.textContent='';

  fetch(MEMORY_GH+path)
    .then(r=>{if(!r.ok)throw new Error('Not found');return r.text()})
    .then(text=>{
      mContent.textContent=text;
      mLines.textContent=text.split('\n').length+' lines';
      mLoading.style.display='none';
      mContent.style.display='block';
    })
    .catch(()=>{
      const p=PROJECTS.find(x=>x.file===path.split('/').pop());
      const desc=p?p.desc:'Memory file';
      mContent.textContent=`# ${title}\n\n${desc}\n\n---\nFile: ${path}\nStatus: Available in local workspace\n\nEdit this file locally or via the agent to add detailed content.`;
      mLines.textContent='\u2014';
      mLoading.style.display='none';
      mContent.style.display='block';
    });
}

function closeModal(){overlay.classList.remove('open');document.body.style.overflow=''}

document.getElementById('modal-close').addEventListener('click',closeModal);
overlay.addEventListener('click',e=>{if(e.target===overlay)closeModal()});
document.addEventListener('keydown',e=>{if(e.key==='Escape')closeModal()});

document.addEventListener('click',e=>{
  const row=e.target.closest('.mem-clickable');
  if(!row)return;
  const name=row.dataset.name;
  const proj=PROJECTS.find(p=>p.file===name);
  openModal(proj?proj.name:name,`projects/${name}`);
});
